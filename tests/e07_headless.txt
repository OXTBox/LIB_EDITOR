--HANDLERS--
-- the card on a stack with the engine stand-ins (LIB_GAMES/tools/engine_stubs); the compile result
function cardOpen pFile
  local tStub
  put textDecode(url ("binfile:/home/igame3d/projects/LIB_GAMES/tools/engine_stubs.livecodescript"), "UTF-8") into tStub
  delete line 1 of tStub
  if there is a stack "EditorExample" then delete stack "EditorExample"
  create stack "EditorExample"
  set the script of stack "EditorExample" to tStub
  set the fileName of stack "EditorExample" to "/home/igame3d/projects/LIB_EDITOR/examples/x.livecode"
  set the script of card 1 of stack "EditorExample" to textDecode(url ("binfile:/home/igame3d/projects/LIB_EDITOR/examples/" & pFile), "UTF-8")
  return the result
end cardOpen

-- Game.OnStart with no keys down, then the card's State
function cardStart pPre
  local tS
  put the long id of card 1 of stack "EditorExample" into tS
  dispatch (pPre & ".TestKeys") to tS with "0"
  dispatch "Game.OnStart" to tS
  dispatch function (pPre & ".State") to tS
  return the result
end cardStart

-- hold pKeys for pSteps steps, then the card's State
function cardRun pPre, pKeys, pSteps
  local tS
  put the long id of card 1 of stack "EditorExample" into tS
  dispatch (pPre & ".TestKeys") to tS with pKeys
  repeat pSteps times
    dispatch "Game.OnUpdate" to tS
  end repeat
  dispatch function (pPre & ".State") to tS
  return the result
end cardRun

-- press and let go of pKeys pTimes times
function cardTap pPre, pKeys, pTimes
  local tS
  if pTimes is empty then put 1 into pTimes
  repeat pTimes times
    get cardRun(pPre, pKeys, 1)
    get cardRun(pPre, "0", 1)
  end repeat
  return it
end cardTap

-- any handler of the card, with up to three parameters
function cardDo pMsg, pA, pB, pC
  dispatch pMsg to card 1 of stack "EditorExample" with pA, pB, pC
  return the result
end cardDo

function cardFn pFn, pA, pB, pC
  dispatch function pFn to card 1 of stack "EditorExample" with pA, pB, pC
  return the result
end cardFn

function ckStart pName, pState, pWant
  if pState begins with pWant then return "PASS " & pName & "   " & pState
  return "FAIL " & pName & " got [" & pState & "] want [" & pWant & "...]"
end ckStart

function ckHas pName, pState, pWant
  if pState contains pWant then return "PASS " & pName & "   " & pState
  return "FAIL " & pName & " got [" & pState & "] want [..." & pWant & "...]"
end ckHas
